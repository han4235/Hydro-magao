const int N = 1e5 + 5;
const int C = 2e3 + 5;
const int oo = 1e9;

int n, q;
int r[N], mark[N], pos[N];
int p[C], cnt;
int d[C][C];
int pre[N], suf[N];

int main() {
    scanf("%d%d", &n, &q);
    for (int i = 1; i <= n; i++) {
        scanf("%d", &r[i]);
        if (r[i])
            mark[i] = mark[r[i]] = 1;
    }
    for (int i = 1; i <= n; i++)
        if (mark[i]) {
            p[++cnt] = i;
            pos[i] = cnt;
        }
    for (int i = 1; i <= cnt; i++) {
        for (int j = 1; j <= i; j++)
            d[i][j] = oo;
        d[i][i] = 0;
        for (int j = i; j > 1; j--) {
            d[i][j - 1] = min(d[i][j - 1], d[i][j] + p[j] - p[j - 1]);
            if (r[p[j]]) {
                int k = pos[r[p[j]]];
                d[i][k] = min(d[i][k], d[i][j] + 1);
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        pre[i] = pre[i - 1];
        if (mark[i])
            pre[i] = i;
    }
    suf[n + 1] = n + 1;
    for (int i = n; i; i--) {
        suf[i] = suf[i + 1];
        if (mark[i])
            suf[i] = i;
    }
    while (q--) {
        int x, y;
        scanf("%d%d", &x, &y);
        if (pre[x] < suf[y])
            printf("%d\n", x - y);
        else
            printf("%d\n", x - pre[x] + d[pos[pre[x]]][pos[suf[y]]] + suf[y] - y);
    }
    return 0;
}

const int N = 105;
const int mod = 1e9;

int n, m;
int f[N][N];
int g[N][N];
int ans;

int main() {
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++)
            f[i][j] = mod;
        f[i][i] = 0;
    }
    for (int i = 1; i <= m; i++) {
        int u, v, w;
        scanf("%d%d%d", &u, &v, &w);
        f[u][v] = min(f[u][v], w);
        f[v][u] = min(f[v][u], w);
    }
    for (int l = 1; l <= n; l++) {
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= n; j++)
                g[i][j] = f[i][j];
        for (int r = l; r <= n; r++) {
            for (int i = 1; i <= n; i++)
                for (int j = 1; j <= n; j++)
                    g[i][j] = min(g[i][j], g[i][r] + g[r][j]);
            for (int i = l; i <= r; i++)
                for (int j = i; j <= r; j++)
                    ans = (ans + g[i][j]) % mod;
        }
    }
    printf("%d\n", ans);
    return 0;
}
