void dfs(int x,int fa){
    dep[x]=dep[fa]+1;
    far[x][col[x]]=dep[x];
    for(int i:g[x]){
        if(i!=fa){
            dfs(i,x);
            for(int j=0;j<2;j++){
                if(far[x][j]!=-1&&far[i][j^1]!=-1){

};

vector<edge> e[maxn];

priority_queue<node, vector<node>, greater<node> > q;
int cnt;
void dijkstra(int s) {
    for(int i=1;i<=cnt;i++)dis[i]=INT_MAX;
    dis[s] = 0;
    q.push(node(0, s));
    while (!q.empty()) {
        int u = q.top().u;
        q.pop();

        if (vis[u]) continue;
        vis[u] = 1;
        for (auto ed : e[u]) {
            int v = ed.v, w = ed.w;
            if (dis[v] > dis[u] + w) {
                dis[v] = dis[u] + w;
                q.push(node(dis[v], v));
            }
        }
    }
}
