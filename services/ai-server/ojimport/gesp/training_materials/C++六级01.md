#include <iostream>
using namespace std;

const int INF = 1000000000;
int cost[2001];
int main() {
    int N = 0, L = 0;

    cin >> N >> L;
    cost[0] = 0;
    for (int i = 1; i <= L; i++)
        cost[i] = INF;
    for (int i = 0; i < N; i++) {
        int c = 0, l = 0;
        cin >> c >> l;
        for (int j = L; j >= 0; j--)
            cost[j] = min(cost[j], cost[max(j - l, 0)] + c);
    }
    if (cost[L] == INF)

        cout << "no solution" << endl;
    else
        cout << cost[L] << endl;
    return 0;
}

#include <iostream>

int num[300000];
int tmp[300000];

long long merge(int l, int r) {
    if (l + 1 == r)
        return 0;
    int m = (l + r) / 2;
    long long res = merge(l, m) + merge(m, r);

    for (int i = l, j = m, k = l; k < r; k++) {
        if (j == r || (i < m && num[i] > num[j])) {
            tmp[k] = num[i];
            i++;

        } else {
            tmp[k] = num[j];
            j++;
            res += m - i;
        }
    }
    for (int k = l; k < r; k++)
        num[k] = tmp[k];
    return res;
}
