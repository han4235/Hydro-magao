int fpw(int b, int e) {
    if (e == 0)
        return 1;
    int r = fpw(b, e >> 1);
    r = 1ll * r * r % p;
    if (e & 1)
        r = 1ll * r * b % p;
    return r;

}

void check(int e) {
    if (fpw(a, e) == 1)
        ans = 0;
}

    int T;
    scanf("%d", &T);
    while (T--) {

        scanf("%d%d", &a, &p);

            if (phi % i == 0) {
                check(phi / i);

                    r /= i;
