int main() {

    int n; cin >> n; assert(1 <= n && n <= 100);
    map<string, int> cnt;
    int mx = -1;
    for (int i = 1; i <= n; i ++) {

        string s; cin >> s;
        assert(s.length() <= 30);
        transform(s.begin(), s.end(), s.begin(), ::tolower);
        if (! cnt.count(s))
            cnt[s] = 0;

    int mx_num = 0;

        if ((it->second) == mx) {

        }
