const int max_n = 1e5 + 10;

int n;
vector<int> edge[max_n];
int val[max_n];

int min_dist[max_n];
int queue[max_n], qh, qt;

void bfs(int src) {
    qh = qt = 0;
    memset(min_dist, 127, sizeof(min_dist));
    queue[++qt] = src;
    min_dist[src] = 0;
    while (qh < qt) {
        int u = queue[++qh];

        for (auto v: edge[u]) {
            if (min_dist[u] + 1 < min_dist[v]) {
                min_dist[v] = min_dist[u] + 1;
                queue[++qt] = v;
            }
        }
    }
}

int main() {
    int m, src, dst;

    ios::sync_with_stdio(false);
    cin >> n >> m >> src >> dst;
    for (int i = 0; i < n; ++i) {
        cin >> val[i];
    }
    for (int i = 0; i < m; ++i) {
        int x, y;
        cin >> x >> y;
        edge[x].push_back(y);
    }

    bfs(src);
    if (min_dist[dst] > n) {
        cout << "No solution\n";
    }
    else {
        cout << min_dist[dst] - val[src] + val[dst] << '\n';
    }

    return 0;
}
