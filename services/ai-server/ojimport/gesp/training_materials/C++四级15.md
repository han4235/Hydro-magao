#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

struct Combo {
    int sum, mx, mn, id;
};

bool cmp(const Combo &a, const Combo &b) {
    if (a.sum != b.sum) return a.sum < b.sum;
    if (a.mx != b.mx) return a.mx < b.mx;
    if (a.mn != b.mn) return a.mn < b.mn;
    return a.id < b.id;
}

int main() {
    int n, k;
    cin >> n >> k;

    vector<Combo> v(n);

    for (int i = 0; i < n; i++) {
        v[i].sum = 0;
        v[i].mx = -1;
        v[i].mn = 1e9;
        v[i].id = i + 1;

        for (int j = 0; j < k; j++) {
            int x;
            cin >> x;
            v[i].sum += x;
            v[i].mx = max(v[i].mx, x);
            v[i].mn = min(v[i].mn, x);
        }
    }

    sort(v.begin(), v.end(), cmp);

    for (int i = 0; i < n; i++) {
        cout << v[i].id;
        if (i + 1 < n) cout << " ";
    }
    cout << endl;

    return 0;
}
