    // freopen("data/1.in", "r", stdin);

    for (int i = 1; i <= m; i ++)
        scanf("%d", &a[i]);

    for (int i = 0; i < n; i ++)
        scanf("%d", &b[i]);

    memset(f, -0x3f, sizeof(f));
    f[0] = 0;
    for (int i = 1; i < n; i ++)
        for (int j = 1; j <= m; j ++)
            if (i - a[j] >= 0)
                f[i] = max(f[i], f[i - a[j]] + b[i - a[j]]);

    int ans = -inf;

    for (int i = 0; i < n; i ++)
        for (int j = 1; j <= m; j ++)
            if (i + a[j] >= n) {
                ans = max(ans, f[i] + b[i]);
                break ;
            }

    cout << ans << endl;
    return 0;
}
